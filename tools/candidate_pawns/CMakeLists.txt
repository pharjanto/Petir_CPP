cmake_minimum_required(VERSION 3.20)
project(PawnPotentialCheck LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENGINE_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../../src" CACHE PATH "Engine source directory")
option(PETIR_CHECK_BOUNDS "Check Pascal array bounds" OFF)
find_package(Threads REQUIRED)
file(GLOB SOURCES "${ENGINE_SRC}/*.cpp")
list(FILTER SOURCES EXCLUDE REGEX "/(main|selftest|ueval)\\.cpp$")
add_executable(evalshow driver.cpp test_hook.cpp ${SOURCES})
target_include_directories(evalshow PRIVATE "${ENGINE_SRC}")
target_link_libraries(evalshow PRIVATE Threads::Threads)
set_target_properties(evalshow PROPERTIES UNITY_BUILD ON UNITY_BUILD_BATCH_SIZE 40)
if(PETIR_CHECK_BOUNDS)
    target_compile_definitions(evalshow PRIVATE PETIR_CHECK_BOUNDS)
endif()
if(MSVC)
    target_compile_options(evalshow PRIVATE /utf-8 /W0 /permissive- /bigobj)
else()
    target_compile_options(evalshow PRIVATE -fwrapv -fno-strict-aliasing)
endif()
