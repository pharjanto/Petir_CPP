cmake_minimum_required(VERSION 3.20)
project(PetirCpp VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Select a 64-bit compiler/kit (x64).")
endif()
option(PETIR_CHECK_BOUNDS "Check the original Pascal array bounds" OFF)
option(PETIR_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
find_package(Threads REQUIRED)
add_executable(petir
    src/header.cpp src/bitboard_mask.cpp src/bitboard.cpp src/udata.cpp
    src/evalmask.cpp src/makemove.cpp src/movgen.cpp src/tools.cpp src/usee.cpp
    src/valid.cpp src/hashing_header.cpp src/hashing.cpp src/repetition.cpp
    src/next.cpp src/ueval.cpp src/search.cpp src/root.cpp src/uiterative.cpp
    src/customboard.cpp src/notation.cpp src/pv.cpp src/opening_book.cpp
    src/etc.cpp src/undoredo.cpp src/engine.cpp src/main.cpp src/selftest.cpp)
target_link_libraries(petir PRIVATE Threads::Threads)
set_target_properties(petir PROPERTIES UNITY_BUILD ON UNITY_BUILD_BATCH_SIZE 32)
if(WIN32)
    set_target_properties(petir PROPERTIES OUTPUT_NAME petir64)
endif()
if(PETIR_CHECK_BOUNDS)
    target_compile_definitions(petir PRIVATE PETIR_CHECK_BOUNDS)
endif()
if(MSVC)
    target_compile_options(petir PRIVATE /utf-8 /W3 /permissive-)
else()
    target_compile_options(petir PRIVATE -fwrapv -fno-strict-aliasing)
    if(PETIR_SANITIZERS)
        target_compile_options(petir PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(petir PRIVATE -fsanitize=address,undefined)
    endif()
endif()
add_custom_command(TARGET petir POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/books" "$<TARGET_FILE_DIR:petir>/books")
enable_testing()
add_test(NAME core-regression COMMAND petir --selftest)
set_tests_properties(core-regression PROPERTIES TIMEOUT 60)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME protocol-regression COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/protocol_test.py" "$<TARGET_FILE:petir>")
    set_tests_properties(protocol-regression PROPERTIES TIMEOUT 60)
endif()
install(TARGETS petir RUNTIME DESTINATION bin)
install(DIRECTORY books DESTINATION bin)
